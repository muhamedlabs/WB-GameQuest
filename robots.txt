# Явно разрешаем ассеты (картинки, стили, скрипты, шрифты) —
# без этого некоторые боты по умолчанию не всегда докачивают их,
# из-за чего превью страницы или картинки могут отображаться в
# поиске некорректно/не полностью.
Allow: /assets/
Allow: *.css
Allow: *.js
Allow: *.webp
Allow: *.svg
Allow: *.png
Allow: *.jpg
Allow: *.mp3
